Quick Start Commands:
---------------------
# 1. Clone the repository
git clone https://github.com/wahidulsami/FullStack_Chatbot_Task_wahidul_sami.git
cd FullStack_Chatbot_Task_wahidul_sami

# 2. Install dependencies
npm --prefix server install
npm --prefix client install

# 3. Configure environment variables (server/.env)
cp .env.example server/.env

# 4. Seed database (optional demo data)
cd server && npm run seed:admin && npm run seed:demo -- --clean

# 5. Run development servers
# Terminal 1: Backend API (port 5000)
cd server && npm run dev

# Terminal 2: Frontend client (port 5173)
cd client && npm run dev
